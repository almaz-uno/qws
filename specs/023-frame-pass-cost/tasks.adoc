== S5x on the desktop of E1

As 020's S5 and its rest variant (020, tasks, "S5 on the desktop of E1"),
with the xterm of `measure.sh` as the window changing. The author away, the
overlay on DP-0 at 144 Hz — the pointer on DP-0 when it starts — the service
stopped meanwhile and started again at the end; a run stops if `i3lock`
comes up, or if `meta` says the window is not of the depth asked. Twelve
runs of S5x, some 70 min (5.3–6.5 min a run in 020), and twelve of the rest
variant, some 30 min, for each build: `develop` (T5) and the change (T10).

[source,bash]
----
# The build of the run: develop with #60 merged (T5), or the change (T10),
# made in a worktree of its own
WT=<the worktree of the build>
make -C "$WT" build && "$WT/qws" version
M=$WT/specs/001-rendering-speed/measure.sh
BIN=$WT/qws
R=~/.cache/qws-runs/$(date +%Y%m%d-%H%M)-023
mkdir -p "$R"
systemctl --user stop qws.service
trap 'xdotool keyup alt 2>/dev/null; systemctl --user start qws.service' EXIT
# one <name> <terminal> <depth> <variables>...
one() {
	local name=$1 term=$2 depth=$3
	shift 3
	if pgrep -x i3lock >/dev/null; then
		echo "i3lock is up: stopping before $name" | tee -a "$R/progress"
		exit 1
	fi
	echo "$(date +%T) start $name" >>"$R/progress"
	env QWS="$BIN" CHANGING=DP-4 CHANGING_TERM="$term" "$@" \
		"$M" run glx "$R/$name" >"$R/$name.out" 2>&1
	echo "$(date +%T) end $name rc=$?" >>"$R/progress"
	if ! grep -q "^changing: DP-4, $term, window [0-9]*, depth $depth\$" "$R/$name/meta"; then
		echo "the window is not a $term of depth $depth: stopping after $name" | tee -a "$R/progress"
		exit 1
	fi
}
# S5x: the xterm, the carousel and the grid, off and on in turn
for i in 1 2 3; do
	for layout in carousel grid; do
		for v in false true; do
			one "s5x-$layout-$v-$i" xterm 24 HOLD=2 LAYOUT=$layout LIVE=$v
		done
	done
done
# The rest variant: the grid, the xterm and the mate-terminal, off and on
for i in 1 2 3; do
	for t in xterm:24 mate-terminal:32; do
		for v in false true; do
			one "rest-${t%:*}-$v-$i" "${t%:*}" "${t#*:}" \
				ACTIVATIONS=10 STEPS=5 HOLD=0 REST=10 LAYOUT=grid LIVE=$v
		done
	done
done
systemctl --user start qws.service
trap - EXIT
for layout in carousel grid; do
	for v in false true; do
		"$M" pool "$R"/s5x-$layout-$v-{1,2,3} >"$R/pool-s5x-$layout-$v"
	done
done
for term in xterm mate-terminal; do
	for v in false true; do
		"$M" pool "$R"/rest-$term-$v-{1,2,3} >"$R/pool-rest-$term-$v"
	done
done
grep -H '^C shown\|^R at rest\|^L live_ms\|^S live\|^A1' "$R"/pool-*
----
