one() {
	local name=$1 bin=$2
	shift 2

		echo "i3lock is up: stopping before $name" | tee -a "$R/progress"
		exit 1

	env QWS="$bin" HOLD=2 CHANGING=DP-4 CHANGING_TERM=mate-terminal "$@" \
		"$M" run glx "$R/$name" >"$R/$name.out" 2>&1

	if ! grep -q '^changing: DP-4, mate-terminal, window [0-9]*, depth 32$' "$R/$name/meta"; then

	fi

for i in 1 2 3; do

# rest <name> <terminal> <depth> <variables>...
rest() {
	local name=$1 term=$2 depth=$3
	shift 3
	if pgrep -x i3lock >/dev/null; then
		echo "i3lock is up: stopping before $name" | tee -a "$R/progress"
		exit 1
	fi
	echo "$(date +%T) start $name" >>"$R/progress"
	env QWS="$BIN" ACTIVATIONS=10 STEPS=5 HOLD=0 REST=10 CHANGING=DP-4 CHANGING_TERM="$term" "$@" \
		"$M" run glx "$R/$name" >"$R/$name.out" 2>&1
	echo "$(date +%T) end $name rc=$?" >>"$R/progress"
	if ! grep -q "^changing: DP-4, $term, window [0-9]*, depth $depth\$" "$R/$name/meta"; then
		echo "the window of S5 is not a $term of depth $depth: stopping after $name" | tee -a "$R/progress"
		exit 1
	fi
}
for i in 1 2; do
	for t in mate-terminal:32 xterm:24; do
		term=${t%:*} depth=${t#*:}
		for layout in carousel grid; do
			rest "rest-$term-$layout-off-$i" "$term" "$depth" LAYOUT=$layout LIVE=false
			rest "rest-$term-$layout-on-$i" "$term" "$depth" LAYOUT=$layout LIVE=true
		done
	done
done
systemctl --user start qws.service
trap - EXIT
for term in mate-terminal xterm; do
	for layout in carousel grid; do
		for v in off on; do
			"$M" pool "$R"/rest-$term-$layout-$v-{1,2} >"$R/pool-$term-$layout-$v"
		done
	done
done
grep -H '^R at rest\|^L live_ms\|^C shown\|^C profile\|missed' "$R"/pool-*

# K9: C of a pool without the glFinish of -v — 100 × (the CPU of C shown
# less the glFinish of C profile) / the time shown — on less off
c() {
	awk '/^C shown/ { cpu = $3; wall = $8 } /^C profile/ { glf = $10 }
		END { printf "%.2f", 100 * (cpu - glf) / wall }' "$1"
}
for term in mate-terminal xterm; do
	for layout in carousel grid; do
		on=$(c "$R/pool-$term-$layout-on") off=$(c "$R/pool-$term-$layout-off")
		echo "K9 $term $layout: $on - $off = $(awk -v a="$on" -v b="$off" 'BEGIN { printf "%.2f", a - b }') points"
	done
done
